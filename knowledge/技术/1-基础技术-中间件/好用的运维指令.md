# 好用的运维指令

查占用某端口的进程
netstat -tnlp | grep <port>
拉取远程binlog
mysqlbinlog -h<实例ip> -P5002 -u<账号> -p<密码> --start-datetime='2018-07-23 19:26:14' --stop-datetime='2018-07-23 19:26:16' --read-from-remote-server 'mysql-bin.000405' --base64-output=decode-rows -v
grep 排除指定类型文件
grep -rn -i "xxxx" ./zc-pos\* \--exclude=\*.{sql,class}
查找并获取某字段数据
grep -r "type:02" pos-router.log | sed 's/\\(.\*\\)\\("serialNo\[^,\]\*\\)\\(.\*\\)/\\2/g' | awk '{stat\[$0\]++} END {for (temp in stat) print temp, stat\[temp\]}'
更简洁的方法：sed 's/\\(.\*\\)type":"\\(\[^"\]\*\\)\\(.\*\\)/\\2/g'
查看连接状态
netstat -o | grep <ip/host>
tcpdump
tcpdump \[-w <cap file>\] -i <network card> <ip/tcp> \[src/dst\] host \[!\] <ip/hostName> and/or \[src/dst\] port <port>
测试网络禁用
规则查看：iptables -L INPUT/OUTPUT/FORWARD --line-numbers \[-n显示端口号\]
禁用ip：iptables -I INPUT/OUTPUT/FORWARD -p all -s <ip> -j DROP
禁用端口：iptables -A INPUT/OUTPUT/FORWARD -p tcp \[-m multiport\] --dport/sport <单端口port/端口范围<port1>:<port2>>|<多个端口<port1>,<port2>,...> -j DROP
使能端口：iptables -A INPUT/OUTPUT/FORWARD -p tcp \[-m multiport\] --dport/sport <单端口port/端口范围<port1>:<port2>>|<多个端口<port1>,<port2>,...> -j ACCEPT
取消某条规则：iptables -D INPUT/OUTPUT/FORWARD <line-number1>,<line-number2>,...
取消所有规则：iptables -F
stdin+file
tail -f client.log | grep "successfully initialized\\|CLOSED" 2>&1 | tee grepClient
截获进程中线程占用资源
top -H -p <pid> -b -n<采样次数> > <output>，结合jstack nid看
tcpdump流量录制回放
录制：tcpdump -i <网卡> port <端口号> -w tcp.dump
mysql流量回放参考：./percona-playback --input-plugin=tcpdump --tcpdump-file=tpcc.dump --tcpdump-mode=fast --tcpdump-port=<端口号> --db-plugin=libmysqlclient --mysql-host=<ip> \--mysql-username=root --mysql-password=<密码> --mysql-port=5002 --mysql-schema=sbtest
